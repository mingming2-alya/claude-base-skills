export async function inferFrame(buf: RingView, budgetMs = 8.4) {
  const frame = buf.borrowZeroCopy(); // zero heap allocation
  const complexity = estimateEntropy(frame);
  const branch = complexity > 0.65 ? "fp16-accurate" : "int8-fast";
  const pred = await runEngine(branch, frame, budgetMs);
  if (pred.confidence < 0.82) return fallbackSafeResult(frame);
  buf.release(frame);
  return pred;
}
